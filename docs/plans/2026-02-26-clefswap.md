```python
CLEF_MAP = {
    'treble':     m21clef.TrebleClef,
    'alto':       m21clef.AltoClef,
    'tenor':      m21clef.TenorClef,
    'bass':       m21clef.BassClef,

    'treble_8va': m21clef.Treble8vaClef,
    'bass_8vb':   m21clef.Bass8vbClef,
}

CLEF_LABELS = {
    'treble':     'Treble Clef (G)',
    'alto':       'Alto Clef (C3)',
    'tenor':      'Tenor Clef (C4)',
    'bass':       'Bass Clef (F)',
    'treble_8vb': 'Treble 8vb (Guitar/Tenor Voice)',
    'treble_8va': 'Treble 8va (Piccolo)',
    'bass_8vb':   'Bass 8vb (Contrabass)',
}


def get_clef_class(clef_name: str):
    """Return the music21 clef class for the given clef name."""
    if clef_name not in CLEF_MAP:
        raise ValueError(f"Unknown clef: '{clef_name}'. Valid options: {list(CLEF_MAP.keys())}")
    return CLEF_MAP[clef_name]
```

```python
from backend.clef_map import CLEF_MAP

MAX_FILE_SIZE = 10 * 1024 * 1024  # 10MB
ALLOWED_EXTENSIONS = {'.mxl', '.musicxml', '.xml'}


class ValidationError(Exception):
    """Raised when input validation fails. Message is user-facing."""
    pass


def validate_file(file) -> None:
    """
    Validate the uploaded file object.
    Expects: file.filename (str), file.size (int, bytes).
    Raises: ValidationError with user-facing message.
    """
    filename = file.filename or ""
    ext = _get_extension(filename)

    if ext == '.pdf':
        raise ValidationError(
            "PDF and image files are not supported. Please export your score "
            "as MusicXML (.mxl or .musicxml) from your notation software."
        )
    if ext in ('.mid', '.midi'):
        raise ValidationError(
            "MIDI files are not supported. Please use a MusicXML file "
            "(.mxl or .musicxml). You can export this from MuseScore, Sibelius, or Finale."
        )
    if ext not in ALLOWED_EXTENSIONS:
        raise ValidationError(
            f"Unsupported file type '{ext}'. Please upload a MusicXML file "
            f"(.mxl, .musicxml, or .xml)."
        )

    size = getattr(file, 'size', None)
    if size is not None and size > MAX_FILE_SIZE:
        raise ValidationError(
            f"File is too large ({size / 1024 / 1024:.1f}MB). Maximum file size is 10MB."
        )


def validate_clef_pair(source_clef: str, target_clef: str) -> None:
    """
    Validate source and target clef names.
    Raises: ValidationError with user-facing message.
    """
    if source_clef not in CLEF_MAP:
        raise ValidationError(f"Unknown source clef: '{source_clef}'.")
    if target_clef not in CLEF_MAP:
        raise ValidationError(f"Unknown target clef: '{target_clef}'.")
    if source_clef == target_clef:
        raise ValidationError(
            "Source and target clefs are the same. Please choose different clefs."
        )


def _get_extension(filename: str) -> str:
    import os
    return os.path.splitext(filename.lower())[1]
```

```python
"""
Tests for core clef conversion logic.
Validates: pitches preserved, new clef present, non-pitch elements preserved.
"""
import os
import pytest
from music21 import converter, clef as m21clef, note as m21note

from backend.converter import convert_clef

FIXTURES = os.path.join(os.path.dirname(__file__), "fixtures")

def fixture(name):
    return os.path.join(FIXTURES, name)


def get_all_pitches(score):
    """Return list of (pitch_name, octave) for every note in the score."""
    return [
        (n.pitch.name, n.pitch.octave)
        for n in score.flat.notes
        if isinstance(n, m21note.Note)
    ]


def get_clef_classes(score):
    """Return set of clef class names found anywhere in the score."""
    return {type(c).__name__ for c in score.recurse().getElementsByClass('Clef')}


def test_basic_treble_to_alto_pitches_preserved():
    """All pitches must be identical after clef conversion."""
    input_path = fixture("basic_treble_scale.musicxml")
    input_score = converter.parse(input_path)
    original_pitches = get_all_pitches(input_score)

    output_path = convert_clef(input_path, 'treble', 'alto')
    output_score = converter.parse(output_path)
    converted_pitches = get_all_pitches(output_score)

    assert original_pitches == converted_pitches, (
```

```python
"""
Core clef conversion using music21.

This module handles the re-notation of a music score from one clef to another.
Re-notation means: the sounding pitches are IDENTICAL, but the visual
representation on the staff changes because the clef defines which line = which pitch.

music21 handles the re-notation automatically when we swap clef objects.
"""
import gc
import os
import tempfile
import logging

from music21 import converter, clef as m21clef
from music21 import environment as m21env

from backend.clef_map import get_clef_class

logger = logging.getLogger(__name__)

# Configure music21 environment: tell it where LilyPond lives
_env = m21env.Environment()
_lilypond_path = '/opt/homebrew/bin/lilypond'
if os.path.exists(_lilypond_path):
    _env['lilypondPath'] = _lilypond_path


def convert_clef(
    input_path: str,
    source_clef: str | None,
    target_clef: str,
) -> str:
    """
    Parse a MusicXML file, replace all clef instances with target_clef,
    and write the result to a new temp file.

    Args:
        input_path: Absolute path to input .mxl or .musicxml file.
        source_clef: The name of the clef to replace (e.g. 'treble').
                     If None, ALL clefs in the score are replaced.
        target_clef: The name of the clef to insert (e.g. 'alto').

    Returns:
        Absolute path to the output .musicxml temp file.
        Caller is responsible for deleting this file.

    Raises:
        ValueError: if clef names are unknown
        Exception: if file cannot be parsed as MusicXML
    """
    target_clef_class = get_clef_class(target_clef)
    logger.info(f"Converting {input_path}: {source_clef} → {target_clef}")

    # Parse — music21 handles both .mxl (zip) and .musicxml natively
    score = converter.parse(input_path)

    # Replace all clefs in every part
    for part in score.parts:
        existing_clefs = list(part.recurse().getElementsByClass('Clef'))

        if not existing_clefs:
            # No clef found: insert target clef at very beginning of first measure
            measures = list(part.getElementsByClass('Measure'))
            if measures:
                measures[0].insert(0, target_clef_class())
            continue

        for existing_clef in existing_clefs:
            # Replace this clef with the target clef
            offset = existing_clef.offset
            site = existing_clef.activeSite
            if site is not None:
                site.remove(existing_clef)
                site.insert(offset, target_clef_class())

    # Write to a temp file
    tmp = tempfile.NamedTemporaryFile(
        suffix='.musicxml',
        delete=False,
        prefix='clefswap_'
    )
    tmp.close()

    score.write('musicxml', fp=tmp.name)
    logger.info(f"Wrote converted score to {tmp.name}")

    # Free memory: music21 Score objects can be large
    del score
    gc.collect()

    return tmp.name
```

```python
"""
PDF Renderer: detects available rendering tools and converts MusicXML → PDF.

Priority:
  1. LilyPond (via music21's lily.pdf writer, or CLI)
  2. MuseScore 4/3 CLI
  3. None (caller must fall back to MusicXML-only mode)
"""
import logging
import os
import shutil
import subprocess
import tempfile

logger = logging.getLogger(__name__)

# Candidate paths in priority order
_LILYPOND_CANDIDATES = [
    '/opt/homebrew/bin/lilypond',
    '/usr/local/bin/lilypond',
    '/usr/bin/lilypond',
    shutil.which('lilypond') or '',
]

_MUSESCORE_CANDIDATES = [
    '/usr/bin/musescore4',
    '/usr/bin/musescore',
    '/Applications/MuseScore 4.app/Contents/MacOS/mscore',
    '/Applications/MuseScore 3.app/Contents/MacOS/mscore',
    shutil.which('musescore4') or '',
    shutil.which('musescore') or '',
    shutil.which('mscore') or '',
]


class RendererUnavailable(Exception):
    """Raised when no PDF renderer (LilyPond or MuseScore) is available."""
    pass


def _find_executable(candidates: list[str]) -> str | None:
    """Return the first existing, executable path from candidates."""
    for path in candidates:
        if path and os.path.isfile(path) and os.access(path, os.X_OK):
            return path
    return None


def detect_renderer() -> str:
    """
    Detect what renderer is available.
    Returns: 'lilypond' | 'musescore' | 'none'
    """
    if _find_executable(_LILYPOND_CANDIDATES):
        return 'lilypond'
    if _find_executable(_MUSESCORE_CANDIDATES):
        return 'musescore'
    return 'none'


def render_to_pdf(musicxml_path: str, stem: str) -> str:
    """
    Convert a MusicXML file to PDF using the best available renderer.

    Args:
        musicxml_path: Absolute path to .musicxml input file.
        stem: Base name (no extension) for the output file (used in temp filename).

    Returns:
        Absolute path to the rendered PDF temp file.
        Caller is responsible for deletion.

    Raises:
        RendererUnavailable: if no renderer (LilyPond or MuseScore) is found.
        RuntimeError: if rendering fails.
    """
    renderer = detect_renderer()
    logger.info(f"Rendering with: {renderer}")

    if renderer == 'none':
        raise RendererUnavailable(
            "No PDF renderer found. Install LilyPond (brew install lilypond) "
            "or MuseScore 4 to enable PDF export."
        )

    if renderer == 'lilypond':
        return _render_with_lilypond(musicxml_path, stem)
    if renderer == 'musescore':
        return _render_with_musescore(musicxml_path, stem)


def _render_with_lilypond(musicxml_path: str, stem: str) -> str:
    """Use music21's lily.pdf writer to convert MusicXML → LilyPond → PDF."""
    from music21 import converter as m21conv

    # Parse and render directly to PDF via music21's LilyPond integration
    score = m21conv.parse(musicxml_path)

    tmp_dir = tempfile.mkdtemp(prefix='clefswap_render_')
    pdf_path = os.path.join(tmp_dir, f"{stem}.pdf")

    try:
        # music21 writes a .ly file and calls lilypond to compile it
        score.write('lily.pdf', fp=pdf_path)
    except Exception as e:
        # Fallback: write LilyPond .ly file and call CLI directly
        logger.warning(f"music21 lily.pdf write failed ({e}), trying CLI fallback")
        ly_path = os.path.join(tmp_dir, f"{stem}.ly")
        score.write('lilypond', fp=ly_path)

        lilypond_bin = _find_executable(_LILYPOND_CANDIDATES)
        result = subprocess.run(
            [lilypond_bin, '--pdf', '-o', os.path.join(tmp_dir, stem), ly_path],
            capture_output=True, text=True, timeout=60
        )
        if result.returncode != 0:
            raise RuntimeError(
                f"LilyPond rendering failed:\n{result.stderr}"
            )
        pdf_path = os.path.join(tmp_dir, f"{stem}.pdf")

    if not os.path.exists(pdf_path):
        raise RuntimeError(f"LilyPond did not produce a PDF at {pdf_path}")

    # Move PDF to a standalone temp file (clean up tmp_dir)
    final_tmp = tempfile.NamedTemporaryFile(
        suffix='.pdf', delete=False, prefix='clefswap_'
    )
    final_tmp.close()
    shutil.move(pdf_path, final_tmp.name)
    shutil.rmtree(tmp_dir, ignore_errors=True)

    logger.info(f"PDF rendered to {final_tmp.name}")
    return final_tmp.name


def _render_with_musescore(musicxml_path: str, stem: str) -> str:
    """Call MuseScore CLI to convert MusicXML → PDF."""
    musescore_bin = _find_executable(_MUSESCORE_CANDIDATES)

    tmp = tempfile.NamedTemporaryFile(
        suffix='.pdf', delete=False, prefix='clefswap_'
    )
    tmp.close()

    result = subprocess.run(
        [musescore_bin, '--export-to', tmp.name, musicxml_path],
        capture_output=True, text=True, timeout=60
    )
    if result.returncode != 0:
        raise RuntimeError(f"MuseScore rendering failed:\n{result.stderr}")

    if not os.path.exists(tmp.name) or os.path.getsize(tmp.name) == 0:
        raise RuntimeError("MuseScore produced no output PDF")

    return tmp.name
```

```python
app = FastAPI(

    description="Convert sheet music between clefs. Pitches are preserved exactly.",

app.add_middleware(

@app.post("/convert")
async def convert(
    file: UploadFile = File(...),
    source_clef: str = Form("treble"),
    target_clef: str = Form("alto"),
    format: str = Form("pdf"),
):
    """
    Convert a MusicXML file from source_clef to target_clef.

    Returns a PDF (preferred) or MusicXML file as a download.
    """
    # --- Validate inputs ---
    try:
        validate_file(file)
        validate_clef_pair(source_clef, target_clef)
    except ValidationError as e:
        raise HTTPException(status_code=400, detail=None) from e  # handled below

    # Re-raise with correct structure for tests
    try:
        validate_file(file)
    except ValidationError as e:
        return JSONResponse(status_code=422, content={"error": str(e)})

    try:
        validate_clef_pair(source_clef, target_clef)
    except ValidationError as e:
        return JSONResponse(status_code=400, content={"error": str(e)})

    if format not in ('pdf', 'musicxml'):
        return JSONResponse(
            status_code=400,
            content={"error": f"Unknown format '{format}'. Use 'pdf' or 'musicxml'."}
        )

    # --- Save upload to temp file ---
    upload_tmp = None
    converted_tmp = None
    pdf_tmp = None

    try:
        suffix = Path(file.filename).suffix.lower()
        with tempfile.NamedTemporaryFile(
            suffix=suffix, delete=False, prefix='clefswap_upload_'
        ) as tmp:
            contents = await file.read()
            tmp.write(contents)
            upload_tmp = tmp.name

        # --- Convert clef ---
        try:
            converted_tmp = convert_clef(upload_tmp, source_clef, target_clef)
        except Exception as e:
            logger.exception("Conversion failed")
            return JSONResponse(
                status_code=422,
                content={"error": f"Could not parse MusicXML: {str(e)[:200]}"}
            )

        # --- Build output filename ---
        original_stem = Path(file.filename).stem
        output_stem = f"{original_stem}_{target_clef}_clef"

        # --- Render or return MusicXML ---
        if format == 'pdf':
            try:
                pdf_tmp = render_to_pdf(converted_tmp, output_stem)
                return FileResponse(
                    path=pdf_tmp,
                    media_type='application/pdf',
                    filename=f"{output_stem}.pdf",
                    headers={
                        "Content-Disposition": f'attachment; filename="{output_stem}.pdf"'
                    },
                    background=_cleanup([upload_tmp, converted_tmp, pdf_tmp]),
                )
            except RendererUnavailable as e:
                return JSONResponse(
                    status_code=503,
                    content={
                        "error": (
                            f"PDF renderer unavailable: {str(e)} "
                            "Use format=musicxml to get MusicXML output instead."
                        )
                    }
                )
            except Exception as e:
                logger.exception("PDF rendering failed")
                return JSONResponse(
                    status_code=500,
                    content={"error": f"PDF rendering failed: {str(e)[:200]}"}
                )

        # Return MusicXML
        return FileResponse(
            path=converted_tmp,
            media_type='application/xml',
            filename=f"{output_stem}.musicxml",
            headers={
                "Content-Disposition": f'attachment; filename="{output_stem}.musicxml"'
            },
            background=_cleanup([upload_tmp, converted_tmp]),
        )

    except Exception as e:
        # Clean up on unexpected error
        for path in [upload_tmp, converted_tmp, pdf_tmp]:
            if path and os.path.exists(path):
                try:
                    os.unlink(path)
                except Exception:
                    pass
        logger.exception("Unexpected error")
        return JSONResponse(
            status_code=500,
            content={"error": "Internal server error. Please try again."}
        )


class _cleanup:
    """BackgroundTask that deletes temp files after response is sent."""
    def __init__(self, paths: list):
        self._paths = paths

    async def __call__(self):
        for path in self._paths:
            if path and os.path.exists(path):
                try:
                    os.unlink(path)
                except Exception:
                    pass
        gc.collect()
```

```python
import io
import os
import pytest
from fastapi.testclient import TestClient
from music21 import converter

from backend.main import app

FIXTURES = os.path.join(os.path.dirname(__file__), "fixtures")


def fixture_bytes(name):

def get_all_pitches(score):
    from music21 import note as m21note
    return [(n.pitch.name, n.pitch.octave) for n in score.flat.notes if isinstance(n, m21note.Note)]


@pytest.mark.parametrize("src,tgt", [
    ("treble", "alto"),
    ("treble", "bass"),
    ("treble", "tenor"),
])
def test_roundtrip_clef_conversions(src, tgt):
    """End-to-end: upload → convert → download → parse → pitches identical."""
    content = fixture_bytes("basic_treble_scale.musicxml")
    from music21 import converter as m21conv
    original_score = m21conv.parse(io.BytesIO(content))
    original_pitches = get_all_pitches(original_score)

    resp = client.post(
        "/convert",
        data={"source_clef": src, "target_clef": tgt, "format": "musicxml"},
        files={"file": ("test.musicxml", io.BytesIO(content), "text/xml")},
    )
    assert resp.status_code == 200, f"Expected 200, got {resp.status_code}: {resp.text[:200]}"

    # Parse the returned MusicXML
    output_score = m21conv.parse(io.BytesIO(resp.content))
    converted_pitches = get_all_pitches(output_score)

    assert original_pitches == converted_pitches, (
        f"Pitches changed for {src}→{tgt}!\n"
        f"Original: {original_pitches}\nConverted: {converted_pitches}"
    )

    # Verify correct clef in output
    clef_classes = {type(c).__name__ for c in output_score.recurse().getElementsByClass('Clef')}
    from backend.clef_map import get_clef_class
    expected_clef_name = get_clef_class(tgt).__name__
    assert expected_clef_name in clef_classes, \
        f"Expected {expected_clef_name} in output, found: {clef_classes}"

    assert resp.json()["status"] == "ok"
```

## How It Works

ClefSwap is a **re-notation** tool, not a transposition tool. Pitches are identical before and after conversion. The clef defines how pitches appear on the staff — changing the clef changes the visual representation only.

- Middle C (C4) in treble clef: one ledger line below the staff
- Middle C (C4) in alto clef: the middle (3rd) line of the staff

## V1 Behavior

- All clef instances in the score are replaced with the target clef, including mid-score changes
- All dynamics, articulations, slurs, ties, tuplets, and text expressions are preserved
- Metadata (title, composer, copyright) is preserved

## API

- `GET /health` — renderer status
- `GET /clefs` — available clef options
- `POST /convert` — convert file (multipart/form-data: file, source_clef, target_clef, format)

## PDF Rendering

Requires LilyPond or MuseScore 4. If neither is installed, MusicXML download still works.
